function updateBulkDeleteState() {
    var count = $('.check-item:checked').length;
    $('#selected-count').text(count);
    if (count > 0) {
        $('#btn-bulk-delete').removeClass('d-none');
    } else {
        $('#btn-bulk-delete').addClass('d-none');
    }
}

function bulkDelete() {
    var selectedIds = [];
    $('.check-item:checked').each(function() {
        selectedIds.push($(this).val());
    });

    if (selectedIds.length === 0) {
        if (typeof Swal !== 'undefined') {
            Swal.fire('Perhatian', 'Pilih minimal satu data keuskupan untuk dihapus.', 'info');
        } else {
            alert('Pilih minimal satu data keuskupan untuk dihapus.');
        }
        return;
    }

    var doDelete = function() {
        var csrfName = $('meta[name="csrf-name"]').attr('content') || '<?= $this->security->get_csrf_token_name(); ?>';
        var csrfHash = $('meta[name="csrf-token"]').attr('content') || '<?= $this->security->get_csrf_hash(); ?>';
        var postData = { ids: selectedIds };
        postData[csrfName] = csrfHash;

        $.ajax({
            url: '<?= site_url("admin/keuskupan/bulk_delete") ?>',
            type: 'POST',
            data: postData,
            dataType: 'json',
            success: function(res) {
                if (res.csrf_hash) {
                    $('meta[name="csrf-token"]').attr('content', res.csrf_hash);
                }
                if (res.success) {
                    if (typeof Swal !== 'undefined') {
                        Swal.fire({
                            title: 'Berhasil!',
                            text: res.message,
                            icon: 'success',
                            confirmButtonColor: '#0061f2'
                        }).then(function() {
                            window.location.reload();
                        });
                    } else {
                        alert(res.message);
                        window.location.reload();
                    }
                } else {
                    if (typeof Swal !== 'undefined') {
                        Swal.fire('Gagal!', res.message, 'error');
                    } else {
                        alert(res.message);
                    }
                }
            },
            error: function() {
                if (typeof Swal !== 'undefined') {
                    Swal.fire('Gagal!', 'Terjadi kesalahan sistem saat menghapus data massal.', 'error');
                } else {
                    alert('Terjadi kesalahan sistem.');
                }
            }
        });
    };

    if (typeof Swal !== 'undefined') {
        Swal.fire({
            title: 'Hapus ' + selectedIds.length + ' Keuskupan?',
            text: 'Apakah Anda yakin ingin menghapus ' + selectedIds.length + ' data keuskupan yang dipilih? Tindakan ini tidak dapat dibatalkan.',
            icon: 'warning',
            showCancelButton: true,
            confirmButtonColor: '#e81500',
            cancelButtonColor: '#69707a',
            confirmButtonText: '<i class="fas fa-trash-alt me-1"></i> Ya, Hapus ' + selectedIds.length + ' Data',
            cancelButtonText: 'Batal',
            customClass: {
                popup: 'shadow-lg rounded-3',
                confirmButton: 'btn btn-danger px-4',
                cancelButton: 'btn btn-secondary px-4'
            },
            buttonsStyling: false
        }).then(function(result) {
            if (result.isConfirmed) {
                doDelete();
            }
        });
    } else {
        if (confirm('Apakah Anda yakin ingin menghapus ' + selectedIds.length + ' data keuskupan terpilih?')) {
            doDelete();
        }
    }
}

function showDetailModal(encryptedId) {
    $('#modal-detail-keuskupan .modal-content').html('<div class="text-center p-5"><div class="spinner-border text-primary" role="status"></div><p class="mt-2 text-muted mb-0">Memuat data detail keuskupan...</p></div>');
    var modalEl = document.getElementById('modal-detail-keuskupan');
    if (window.bootstrap && bootstrap.Modal) {
        var myModal = bootstrap.Modal.getOrCreateInstance(modalEl);
        myModal.show();
    } else {
        $('#modal-detail-keuskupan').modal('show');
    }
    
    $.ajax({
        url: '<?= site_url("admin/keuskupan/detail_ajax") ?>/' + encryptedId,
        type: 'GET',
        success: function(response) {
            $('#modal-detail-keuskupan .modal-content').html(response);
            try {
                if (typeof feather !== 'undefined') feather.replace();
            } catch(e) {}
        },
        error: function() {
            $('#modal-detail-keuskupan .modal-content').html('<div class="modal-body p-4 text-center text-danger"><i class="fas fa-exclamation-circle fa-2x mb-2"></i><p class="mb-0">Gagal memuat detail data keuskupan.</p></div>');
        }
    });
}

function konfirmasiHapus(url, nama) {
    if (typeof Swal !== 'undefined') {
        Swal.fire({
            title: 'Hapus Keuskupan?',
            text: 'Apakah Anda yakin ingin menghapus data keuskupan "' + nama + '"? Tindakan ini tidak dapat dibatalkan.',
            icon: 'warning',
            showCancelButton: true,
            confirmButtonColor: '#e81500',
            cancelButtonColor: '#69707a',
            confirmButtonText: '<i class="fas fa-trash-alt me-1"></i> Ya, Hapus',
            cancelButtonText: 'Batal',
            customClass: {
                popup: 'shadow-lg rounded-3',
                confirmButton: 'btn btn-danger px-4',
                cancelButton: 'btn btn-secondary px-4'
            },
            buttonsStyling: false
        }).then(function(result) {
            if (result.isConfirmed) {
                window.location.href = url;
            }
        });
    } else {
        if (confirm('Apakah Anda yakin ingin menghapus keuskupan ' + nama + '?')) {
            window.location.href = url;
        }
    }
}

    public function auto_seed_keuskupan_if_empty()
    {
        if (!$this->db->table_exists($this->table)) return;

        $has_is_deleted = $this->db->field_exists('is_deleted', $this->table);
        $has_status     = $this->db->field_exists('status', $this->table);
        $has_created_at = $this->db->field_exists('created_at', $this->table);

        $default_list = $this->get_default_keuskupan_names();

        // Ambil nama-nama yang sudah ada di database
        $existing_names = [];
        $query = $this->db->select('nama_keuskupan')->get($this->table);
        if ($query && $query->num_rows() > 0) {
            foreach ($query->result() as $row) {
                if (!empty($row->nama_keuskupan)) {
                    $existing_names[strtolower(trim($row->nama_keuskupan))] = true;
                }
            }
        }

        // Insert keuskupan yang belum ada
        foreach ($default_list as $nama) {
            if (!isset($existing_names[strtolower(trim($nama))])) {
                $insert = ['nama_keuskupan' => $nama];
                if ($has_status)     $insert['status'] = 'Aktif';
                if ($has_is_deleted) $insert['is_deleted'] = 0;
                if ($has_created_at) $insert['created_at'] = date('Y-m-d H:i:s');
                $this->db->insert($this->table, $insert);
                $existing_names[strtolower(trim($nama))] = true;
            }
        }

            $this->db->group_start()
                     ->where('keuskupan.is_deleted', 0)
                     ->or_where('keuskupan.is_deleted IS NULL', NULL, FALSE)
                     ->group_end();
        }

        $row = $this->db->get()->row();

        if (!$row) {
            $this->db->from($this->table);
            $this->db->where($pk, $id);
            if ($has_is_deleted) {
                $this->db->group_start()
                         ->where('is_deleted', 0)
                         ->or_where('is_deleted IS NULL', NULL, FALSE)
                         ->group_end();
            }
            $row = $this->db->get()->row();
        }

        return $row;
    }

    public function insert($data)
    {
        $this->ensure_table_schema();
        return $this->db->insert($this->table, $data);
    }

    public function update($id, $data)
    {
        $this->ensure_table_schema();
        $pk = $this->db->field_exists('id_keuskupan', $this->table) ? 'id_keuskupan' : 'id';
        $this->db->where($pk, $id);
        return $this->db->update($this->table, $data);
    }

    public function delete($id)
    {
        $this->ensure_table_schema();
        $pk = $this->db->field_exists('id_keuskupan', $this->table) ? 'id_keuskupan' : 'id';
        $this->db->where($pk, $id);
        return $this->db->update($this->table, ['is_deleted' => 1, 'deleted_at' => date('Y-m-d H:i:s')]);
    }

    public function delete_many($ids)
    {
        $this->ensure_table_schema();
        $ids = array_values(array_unique(array_filter(array_map('intval', (array) $ids))));
        if (empty($ids)) {
            return false;
        }

        $pk = $this->db->field_exists('id_keuskupan', $this->table) ? 'id_keuskupan' : 'id';
        $this->db->where_in($pk, $ids);
        return $this->db->update($this->table, [
            'is_deleted' => 1,
            'deleted_at' => date('Y-m-d H:i:s'),
        ]);
    }
